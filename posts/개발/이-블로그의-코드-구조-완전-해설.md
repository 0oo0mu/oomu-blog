# 이 블로그는 어떻게 만들었나 — 코드 완전 해설

이 블로그는 React나 Next.js 같은 프레임워크 없이, 순수 JavaScript(ES Modules)로만 만들어졌습니다.
파일 구조는 `core/`(공통 기반)와 `modules/`(기능 단위)로 나뉩니다.

```
js/
├── core/
│   ├── app.js        ← 이벤트 버스 + 모듈 레지스트리
│   ├── router.js     ← SPA 라우터
│   ├── config.js     ← 사이트 설정 (자동 생성)
│   ├── theme.js      ← 다크/라이트 테마
│   ├── accent.js     ← 컬러 테마 (8가지 프리셋)
│   ├── storage.js    ← localStorage 래퍼
│   └── env.js        ← 로컬/웹 환경 감지
└── modules/
    ├── posts-loader.js ← posts.json 로더
    ├── filter.js       ← 카테고리·태그·검색 통합 필터
    ├── search.js       ← 실시간 검색
    ├── renderer.js     ← 카드 렌더러
    ├── sidebar.js      ← 카테고리 트리 사이드바
    ├── toc.js          ← 목차(ToC) 생성
    ├── music-player.js ← 하단 뮤직 플레이어
    ├── editor.js       ← 마크다운 에디터
    └── markdown.js     ← 마크다운 파서 래퍼
```

## Core 레이어

### `js/core/app.js` — 이벤트 버스 & 모듈 레지스트리

모든 모듈이 서로 직접 `import`하지 않고, `App`을 통해 이벤트로 소통합니다.
이 방식을 **Pub/Sub 패턴**이라고 부릅니다. 모듈 간 결합도가 낮아져서
나중에 기능을 추가하거나 제거해도 다른 파일을 건드리지 않아도 됩니다.

```js
const App = {
  modules: {},      // 등록된 모듈들
  _listeners: {},   // 이벤트 리스너들

  // 모듈 등록 + init() 자동 호출
  register(name, module) {
    this.modules[name] = module;
    if (typeof module.init === 'function') {
      module.init();
    }
  },

  // 이벤트 발행: 구독한 모든 콜백 즉시 실행
  emit(event, data) {
    const listeners = this._listeners[event] || [];
    listeners.forEach(cb => {
      try { cb(data); } catch (e) {
        console.error(`[App] 이벤트 핸들러 오류 (${event}):`, e);
      }
    });
  },

  // 이벤트 구독, 반환값으로 구독 해제 가능
  on(event, cb) {
    if (!this._listeners[event]) this._listeners[event] = [];
    this._listeners[event].push(cb);
    return () => {
      this._listeners[event] = this._listeners[event].filter(fn => fn !== cb);
    };
  },

  get(name) { return this.modules[name]; },
};
```

**현재 사용 중인 이벤트 목록:**

| 이벤트 | 발행자 | 의미 |
|---|---|---|
| `posts:loaded` | posts-loader | 포스트 데이터 로드 완료 |
| `posts:filtered` | filter | 필터링된 포스트 목록 |
| `filter:category` | sidebar | 카테고리 선택 |
| `filter:search` | search | 검색어 입력 |
| `theme:change` | theme | 테마 전환 |
| `router:post` | router | 포스트 뷰로 전환 |
| `router:list` | router | 목록 뷰로 전환 |

### `js/core/router.js` — SPA 라우터

**왜 SPA인가?** 멀티 페이지(index.html → post.html) 방식은 페이지 이동마다 브라우저가 전체를 새로 로드합니다. 이때 `<audio>` 엘리먼트도 파괴되어 **음악이 끊깁니다.**

SPA는 실제 페이지 이동 없이 콘텐츠만 바꿔치기합니다. 뮤직 플레이어, 사이드바, 헤더가 그대로 살아있으므로 음악이 끊기지 않습니다.

```js
const Router = {
  _savedScrollY: 0,  // 뒤로가기 시 스크롤 위치 복원용

  init() {
    // 직접 URL 접근 처리: ?file=... 이면 포스트 뷰로 시작
    const params = new URLSearchParams(window.location.search);
    const file = params.get('file');
    if (file) {
      this._applyView('post', { file }, false);
    } else {
      this._applyView('list', {}, false);
    }

    // 이벤트 위임: 문서 전체에 한 번만 달아서
    // 동적으로 생성되는 카드에도 자동으로 적용됨
    document.addEventListener('click', (e) => {
      const card = e.target.closest('.post-card[data-file]');
      if (card) {
        e.preventDefault();
        this.goPost(card.dataset.file);
        return;
      }
      if (e.target.closest('#backBtn')) {
        e.preventDefault();
        this.goList();
      }
    });

    // 브라우저 뒤로/앞으로 버튼
    window.addEventListener('popstate', (e) => {
      const state = e.state || { view: 'list' };
      this._applyView(state.view, state, false);
    });
  },

  goPost(file) {
    this._savedScrollY = window.scrollY;  // 현재 스크롤 위치 저장
    history.pushState({ view: 'post', file }, '', `?file=${encodeURIComponent(file)}`);
    this._applyView('post', { file }, true);
  },

  goList() {
    history.pushState({ view: 'list' }, '', 'index.html');
    this._applyView('list', {}, true);
  },

  _applyView(view, data, scroll) {
    const listEl = document.getElementById('listView');
    const postEl = document.getElementById('postView');

    if (view === 'post') {
      listEl?.classList.add('view-hidden');
      postEl?.classList.remove('view-hidden');
      App.emit('router:post', { file: data.file });
      if (scroll) window.scrollTo({ top: 0, behavior: 'instant' });
    } else {
      postEl?.classList.add('view-hidden');
      listEl?.classList.remove('view-hidden');
      App.emit('router:list', {});
      if (scroll) {
        // requestAnimationFrame: DOM이 표시된 후에 스크롤해야 정확함
        requestAnimationFrame(() => {
          window.scrollTo({ top: this._savedScrollY, behavior: 'instant' });
        });
      }
    }
  },
};
```

**동작 흐름:**
1. 카드 클릭 → `goPost(file)` → URL을 `?file=...`로 변경 (페이지 이동 없음)
2. `#listView` 숨기고 `#postView` 표시 → `App.emit('router:post')`
3. 뒤로가기 → `popstate` 이벤트 → `goList()`

### `js/core/storage.js` — localStorage 래퍼

localStorage를 직접 쓰면 try/catch가 매번 필요하고 흩어지기 쉽습니다.
이 모듈에서 한 곳에 모아서 관리합니다.

```js
const Storage = {
  set(key, value) {
    try {
      localStorage.setItem(key, JSON.stringify(value));
    } catch (e) {
      console.warn('[Storage] 저장 실패:', key, e);
    }
  },

  get(key, defaultValue = null) {
    try {
      const raw = localStorage.getItem(key);
      // null이면 키가 없다는 뜻 → 기본값 반환
      return raw !== null ? JSON.parse(raw) : defaultValue;
    } catch (e) {
      console.warn('[Storage] 읽기 실패:', key, e);
      return defaultValue;
    }
  },

  remove(key) {
    try {
      localStorage.removeItem(key);
    } catch (e) {
      console.warn('[Storage] 삭제 실패:', key, e);
    }
  },
};
```

값은 자동으로 `JSON.stringify` / `JSON.parse`하므로 숫자, 배열, 객체도 그대로 저장하고 꺼낼 수 있습니다.

### `js/core/theme.js` — 다크/라이트 테마

```js
const Theme = {
  init() {
    const saved = Storage.get('theme', 'light');
    this._apply(saved);
    const btn = document.getElementById('themeToggle');
    if (btn) btn.addEventListener('click', () => this.toggle());
  },

  toggle() {
    const next = document.documentElement.classList.contains('dark') ? 'light' : 'dark';
    this._apply(next);
    Storage.set('theme', next);
    App.emit('theme:change', { theme: next });
  },

  _apply(theme) {
    if (theme === 'dark') {
      document.documentElement.classList.add('dark');
      this._setButton('☀️', '라이트');
    } else {
      document.documentElement.classList.remove('dark');
      this._setButton('🌙', '다크');
    }
  },
};
```

**핵심 설계 포인트:** `dark` 클래스를 `<body>`가 아닌 `<html>` 태그(`document.documentElement`)에 붙입니다. 그 이유는 **섬광 방지** 때문입니다.

각 HTML 파일의 `<head>`에 인라인 스크립트가 있는데, 이 스크립트가 CSS 로드보다 먼저 실행되며 다크 클래스를 붙입니다. 이 시점에 `document.body`는 아직 null이지만, `document.documentElement`는 항상 사용 가능합니다.

### `js/core/accent.js` — 컬러 테마 (8가지 프리셋)

자유 색상 선택 대신 8가지 미리 정의된 테마를 제공합니다.
각 프리셋은 `--accent`(포인트 컬러)와 `--accent-light`(배경용 연한 색)를 함께 정의하므로 사이드바, 카드, 버튼 등 모든 곳에서 일관되게 보입니다.

```js
const PRESETS = [
  { name: '인디고', accent: '#6366f1', light: 'rgba(99,102,241,0.13)'  },
  { name: '블루',   accent: '#3b82f6', light: 'rgba(59,130,246,0.13)'  },
  { name: '에메랄드', accent: '#10b981', light: 'rgba(16,185,129,0.13)' },
  // ... 5가지 더
];

const Accent = {
  _current: 0,

  init() {
    this._current = Storage.get('accent_preset', 0);
    this._apply(this._current);
    this._buildPopup();   // 색상 선택 팝업 생성
    this._bindEvents();   // 클릭 이벤트 연결

  _apply(index) {
    const preset = PRESETS[index];
    // CSS 변수를 직접 교체 → 모든 색상이 한 번에 바뀜
    document.documentElement.style.setProperty('--accent',       preset.accent);
    document.documentElement.style.setProperty('--accent-light', preset.light);

    document.querySelectorAll('.color-dot').forEach(dot => {
      dot.style.background = preset.accent;
    });
  },
};
```

### `js/modules/posts-loader.js` — 포스트 데이터 로더

`posts/posts.json`을 `fetch`로 가져와서 날짜순 정렬 후 `App` 이벤트로 알립니다.

```js
let _cache = null;  // 한 번 불러온 데이터는 재요청하지 않음

const PostsLoader = {
  async load() {
    if (_cache) {
      App.emit('posts:loaded', { posts: _cache });
      return _cache;
    }

    const res = await fetch('posts/posts.json');
    if (!res.ok) throw new Error(`HTTP ${res.status}`);

    const posts = await res.json();

    // 최신 글이 먼저 오도록 날짜 내림차순 정렬
    posts.sort((a, b) => new Date(b.date) - new Date(a.date));

    _cache = posts;
    App.emit('posts:loaded', { posts });
    return posts;
  },
};
```

모듈 스코프 변수 `_cache`로 캐싱합니다. 한 세션 안에서 SPA로 목록을 여러 번 보더라도 네트워크 요청은 딱 한 번만 합니다.

### `js/modules/filter.js` — 통합 필터

카테고리 + 태그 + 검색어, 세 가지 필터를 **AND 조건**으로 합칩니다.

```js
const Filter = {
  state: {
    category: 'all',  // 'all' 또는 카테고리 경로
    tag:      'all',  // 'all' 또는 태그 이름
    query:    '',     // 검색어 (빈 문자열이면 전체)
  },

  init() {
    App.on('posts:loaded', ({ posts }) => {
      this._allPosts = posts;
      this._renderTagChips();
      this._applyFilter();
    });
    App.on('filter:category', ({ category }) => {
      this.state.category = category;
      this._applyFilter();
    });
    App.on('filter:search', ({ query }) => {
      this.state.query = query;
      this._applyFilter();
    });
  },

  _applyFilter() {
    const { category, tag, query } = this.state;

    const filtered = this._allPosts.filter(post => {
      // 카테고리 계층 필터: 'PC' → 'PC', 'PC/언어', 'PC/언어/C' 모두 포함
      const catOk = category === 'all'
        || post.category === category
        || post.category?.startsWith(category + '/');

      const tagOk = tag === 'all' || (post.tags || []).includes(tag);
      const searchOk = !query || this._matchSearch(post, query);

      return catOk && tagOk && searchOk;
    });

    App.emit('posts:filtered', { posts: filtered, query });
  },

  _matchSearch(post, query) {
    const text = [post.title, post.excerpt, post.category, ...(post.tags || [])]
      .join(' ').toLowerCase();

    // 여러 단어를 AND로 검색: "javascript 입문" → 두 단어 모두 있어야 매칭
    const words = query.toLowerCase().split(/\s+/).filter(Boolean);
    return words.every(word => text.includes(word));
  },
};
```

### `js/modules/search.js` — 실시간 검색

사용자 입력을 감지하고 `filter:search` 이벤트를 발행합니다.

```js
const Search = {
  init() {
    const input    = document.getElementById('searchInput');
    const clearBtn = document.getElementById('searchClear');

    let debounce = null;

    input.addEventListener('input', () => {
      clearTimeout(debounce);
      // 타이핑이 멈추고 200ms 후에 검색 실행 (너무 잦은 필터링 방지)
      debounce = setTimeout(() => {
        const query = input.value.trim();
        if (clearBtn) clearBtn.style.display = query ? 'flex' : 'none';
        App.emit('filter:search', { query });
      }, 200);
    });

    // Escape 키로 검색 초기화
    input.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        input.value = '';
        App.emit('filter:search', { query: '' });
        input.blur();
      }
    });

    // '/' 키로 검색창 포커스 (타이핑 중엔 동작 안 함)
    document.addEventListener('keydown', (e) => {
      const tag = document.activeElement?.tagName;
      const isTyping = ['INPUT', 'TEXTAREA', 'SELECT'].includes(tag);
      if (e.key === '/' && !isTyping) {
        e.preventDefault();
        input.focus();
        input.select();
      }
    });
  },
};
```

**디바운스(Debounce):** 키를 누를 때마다 즉시 필터링하면 너무 자주 실행됩니다. `setTimeout`으로 200ms를 기다렸다가, 그 사이 새 입력이 오면 타이머를 리셋합니다. 타이핑이 멈춰야 검색이 실행됩니다.

### `js/modules/renderer.js` — 포스트 카드 렌더러

`posts:filtered` 이벤트를 받아 카드 목록을 DOM에 그립니다.
검색 중이면 검색어를 `<mark>` 태그로 하이라이트합니다.

```js
const Renderer = {
  init() {
    App.on('posts:filtered', ({ posts, query = '' }) => {
      this.render(posts, query);
    });
  },

  render(posts, query = '') {
    const grid = document.getElementById('postsGrid');
    if (posts.length === 0) {
      grid.innerHTML = `<div class="empty-state">...</div>`;
      return;
    }
    grid.innerHTML = posts.map(post => this._cardHTML(post, query)).join('');
  },

  _cardHTML(post, query) {
    // data-file 속성 ← SPA 라우터의 핵심!
    // router.js가 .post-card[data-file] 클릭을 가로채서 페이지 이동 없이 처리
    const href = `post.html?file=${encodeURIComponent(post.file)}`;  // JS 없는 환경 폴백

    const title   = query ? highlight(escapeHtml(post.title), query)   : escapeHtml(post.title);
    const excerpt = query ? highlight(escapeHtml(post.excerpt), query)  : escapeHtml(post.excerpt);

    return `
      <a class="post-card" href="${href}" data-file="${escapeHtml(post.file)}">
        <span class="post-category">${escapeHtml(post.category)}</span>
        <h2 class="post-card-title">${title}</h2>
        <p class="post-card-excerpt">${excerpt}</p>
        <div class="post-card-meta">
          <span class="post-date">${formatDate(post.date)}</span>
          <div class="post-tags">${(post.tags || []).map(t => `<span>#${t}</span>`).join('')}</div>
        </div>
      </a>`;
  },
};

// 검색어를 <mark>로 감싸서 하이라이트
function highlight(text, query) {
  const words = query.trim().split(/\s+/).filter(Boolean);
  const pattern = words.map(w => w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')).join('|');
  return text.replace(new RegExp(`(${pattern})`, 'gi'), '<mark>$1</mark>');
}
```

### `js/modules/sidebar.js` — 카테고리 트리 사이드바

`posts.json`의 `category` 필드(예: `"PC/언어/C"`)를 파싱해서 폴더 트리를 만듭니다.

```js
// "PC/언어/C"를 트리 구조로 변환하는 과정

// 트리 노드 구조
// { _posts: 직접 포스트 수, _total: 하위 포함 총합, _children: { 이름: 노드 } }

posts.forEach(post => {
  const parts = post.category.split('/');  // ['PC', '언어', 'C']
  let node = root;

  parts.forEach((part, depth) => {
    if (!node._children[part]) {
      node._children[part] = { _posts: 0, _children: {} };
    }
    // 마지막 파트에만 직접 포스트 수 +1
    if (depth === parts.length - 1) {
      node._children[part]._posts++;
    }
    node = node._children[part];
  });
});

// 하위 포스트 합산 (재귀)
function calcTotal(node) {
  const childrenTotal = Object.values(node._children)
    .reduce((sum, child) => sum + calcTotal(child), 0);
  node._total = node._posts + childrenTotal;
  return node._total;
}
```

**폴더 열기/닫기 버그 수정 포인트:**
클릭 핸들러에서 `toggle()`로 열고 닫는데, 기존 코드는 `_select()` 내부에서 선택된 노드 자신을 포함해 전체 경로를 강제로 `open` 처리했습니다. 그래서 닫으려고 클릭해도 즉시 다시 열렸습니다.

수정: `parts.slice(0, -1)` — 자기 자신(마지막 요소)을 제외하고 조상만 열어줍니다.

```js
// 수정 전: 자신 포함 전체 경로 열기
parts.forEach(part => { ... node.classList.add('open') ... });

// 수정 후: 조상만 열기 (자신은 toggle 결과 유지)
const ancestors = parts.slice(0, -1);
ancestors.forEach(part => { ... node.classList.add('open') ... });
```

### `js/modules/toc.js` — 목차 생성 & 스크롤 감지

포스트 본문의 `h2`, `h3`, `h4` 태그를 찾아서 목차를 만들고, 스크롤 위치에 따라 현재 섹션을 하이라이트합니다.

```js
const Toc = {
  build({ bodyId = 'postBody', tocId = 'tocList', sidebarId = 'tocSidebar' } = {}) {
    const headings = Array.from(body.querySelectorAll('h2, h3, h4'));

    // 각 heading에 id 부여
    headings.forEach((heading, idx) => {
      const id = `h-${idx}-` + heading.textContent
        .toLowerCase()
        .replace(/[^\w\s가-힣]/g, '')  // 특수문자 제거
        .replace(/\s+/g, '-')           // 공백을 하이픈으로
        .slice(0, 40);
      heading.id = id;

      // 목차 항목 생성 (level-2, level-3, level-4 클래스로 들여쓰기)
      const li = document.createElement('li');
      li.className = `toc-item level-${parseInt(heading.tagName[1])}`;
      // ...
    });

    // 스크롤 이벤트로 현재 섹션 감지
    // requestAnimationFrame으로 프레임당 한 번만 실행 (성능 최적화)
    this._scrollHandler = () => {
      if (this._rafId) cancelAnimationFrame(this._rafId);
      this._rafId = requestAnimationFrame(() => {
        this._updateActive(tocList);
      });
    };
    window.addEventListener('scroll', this._scrollHandler, { passive: true });
  },

  _updateActive(tocList) {
    // 헤더 높이 + 16px 아래를 기준선으로
    const threshold = window.scrollY + headerHeight + 16;

    // 기준선보다 위에 있는(= 이미 지나친) heading 중 가장 마지막 것 = 현재 섹션
    let activeId = null;
    for (const heading of this._headings) {
      if (heading.offsetTop <= threshold) {
        activeId = heading.id;
      } else {
        break;
      }
    }
    if (activeId) this._setActive(tocList, activeId);
  },
};
```

**왜 IntersectionObserver를 쓰지 않았나?**
IntersectionObserver는 heading이 "화면에 들어올 때"만 감지합니다. 섹션이 길면 heading이 위로 사라진 뒤 하이라이트가 꺼집니다. 스크롤 이벤트 + 오프셋 계산 방식이 더 안정적입니다.

### `js/modules/music-player.js` — 뮤직 플레이어

페이지 이동 시 음악이 끊기지 않도록 SPA와 연동된 뮤직 플레이어입니다.

**주요 기능:** 재생/일시정지, 이전/다음 곡, 볼륨, 셔플, 반복 모드(없음/한 곡/전체), 플레이리스트 패널, 페이지 이동 후 재생 위치 복원.

```js
  _prev() {
```

### `js/modules/markdown.js` — 마크다운 파서

CDN으로 불러온 `marked.js`를 래핑합니다.
Front Matter(파일 맨 위 `---` 구간) 파싱도 여기서 담당합니다.

```js
const Markdown = {
  parse(mdText) {
    return window.marked.parse(mdText);
  },

  // Front Matter 파싱
  // ---
  // title: 내 첫 글
  // date: 2024-01-15
  // tags: [JS, CSS]
  // ---
  // # 본문
  parseFrontMatter(raw) {
    const meta = {};
    const match = raw.match(/^---\n([\s\S]*?)\n---\n([\s\S]*)$/);
    if (!match) return { meta, content: raw };

    match[1].split('\n').forEach(line => {
      const colonIdx = line.indexOf(':');
      if (colonIdx === -1) return;
      const key = line.slice(0, colonIdx).trim();
      const val = line.slice(colonIdx + 1).trim();

      if (key === 'tags') {
        // [JS, CSS] 또는 JS, CSS 형식 모두 지원
        meta.tags = val
          .replace(/[\[\]]/g, '')
          .split(',')
          .map(t => t.trim())
          .filter(Boolean);
      } else {
        meta[key] = val;
      }
    });

    return { meta, content: match[2] };
  },
};
```

### `build.js` — 배포 전 빌드

`node build.js` (또는 `npm run build`)를 실행하면 다음 작업이 순서대로 이루어집니다:

1. `blog.config.json` 읽기
2. `posts/` 폴더를 재귀 탐색해서 `.md` 파일 수집
3. 각 파일의 Front Matter 파싱
4. `posts/posts.json` 생성 (목록 데이터)
5. `sitemap.xml` 생성 (SEO)
6. `robots.txt` 생성 (SEO)
7. `js/core/config.js` 생성 (프론트엔드용 설정 파일)

```js
// posts/ 폴더 재귀 탐색
function findMarkdownFiles(dir, baseDir = dir) {
  const entries = fs.readdirSync(dir, { withFileTypes: true });
  const files = [];

  for (const entry of entries) {
    const fullPath = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      files.push(...findMarkdownFiles(fullPath, baseDir));  // 재귀
    } else if (entry.isFile() && entry.name.endsWith('.md')) {
      // Windows 경로 구분자(\)를 슬래시(/)로 통일
      const rel = path.relative(baseDir, fullPath).replace(/\\/g, '/');
      files.push(rel);
    }
  }
  return files;
}
```

새 글을 쓴 후에는 반드시 `node build.js`를 실행해야 `posts.json`이 업데이트됩니다.
`js/core/config.js`는 직접 수정하지 말고, `blog.config.json`을 수정 후 빌드하세요.
